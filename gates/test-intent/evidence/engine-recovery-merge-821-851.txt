// Temporary cross-feature probe; not part of the submitted test suite.
#[cfg(test)]
mod recovery_merge_probe {
    use super::*;
    #[test]
    fn recovery_evidence_and_explicit_reset_keep_their_separate_lifetimes() {
        let root = tempfile::Builder::new().prefix("recovery reset path ").tempdir().unwrap();
        let dir = root.path();
        crate::recovery::jvm_options(dir).unwrap();
        let fatal = dir.join("crashes/hs_err_pid-first.log");
        let heap = dir.join("crashes/heap-first.hprof");
        fs::write(&fatal, "#  Internal Error (fixture.cpp:1)\n").unwrap();
        fs::write(&heap, b"first crash evidence").unwrap();
        for keep in ["models", "native-bin/llama-server"] {
            fs::create_dir_all(dir.join(keep)).unwrap();
            fs::write(dir.join(keep).join("keep"), b"retained").unwrap();
        }
        fs::write(dir.join("search-shortcut.v1.json"), br#"{"version":1,"enabled":false}"#).unwrap();
        let idle = run_pending(dir, || panic!("normal recovery must not perform reset cleanup"));
        crate::after_factory_reset(&idle, || { crate::recovery::jvm_options(dir).unwrap(); });
        assert!(fatal.is_file() && heap.is_file());
        request(dir).unwrap();
        let awaiting = run_pending(dir, || Ok(()));
        assert_eq!(awaiting.phase, Phase::AwaitingStorage);
        crate::after_factory_reset(&awaiting, || panic!("blocked reset must not recreate crash directories or start supervision"));
        assert!(!dir.join("crashes").exists());
        assert!(!dir.join("search-shortcut.v1.json").exists());
        storage_cleared(dir).unwrap();
        let complete = run_pending(dir, || panic!("a completed reset must not repeat deletion"));
        assert_eq!(complete.phase, Phase::Complete);
        crate::after_factory_reset(&complete, || { crate::recovery::jvm_options(dir).unwrap(); });
        assert!(dir.join("crashes").is_dir());
        assert!(!fatal.exists() && !heap.exists());
        for keep in ["models", "native-bin/llama-server"] {
            assert_eq!(fs::read(dir.join(keep).join("keep")).unwrap(), b"retained");
        }
    }
}
